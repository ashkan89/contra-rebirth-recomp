cmake_minimum_required(VERSION 3.16)
project(chocorecomp CXX)
option(USE_SDL "Build with a window, audio and controller input on Linux" OFF)
option(PI_ZERO2 "Use conservative defaults for experimental Pi Zero 2 W builds" OFF)
option(USE_GLES2 "Use the hardware OpenGL ES 2.0 renderer" OFF)
if(USE_GLES2 AND (WIN32 OR NOT USE_SDL))
  message(FATAL_ERROR "USE_GLES2 requires the Linux SDL platform (-DUSE_SDL=ON).")
endif()
if(WIN32 AND NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows builds require a 64-bit toolchain.")
endif()
if(PI_ZERO2 AND (NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT USE_SDL OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64|armv7l|armv8l|arm)$"))
  message(FATAL_ERROR "PI_ZERO2 requires Linux ARM and -DUSE_SDL=ON.")
endif()
if(WIN32)
  enable_language(RC)
endif()
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()
file(GLOB GEN_SRC gen/*.cpp)
file(GLOB RT_SRC runtime/*.cpp)
add_executable(contrarebirth ${RT_SRC} ${GEN_SRC})
target_include_directories(contrarebirth PRIVATE runtime gen)
if(PI_ZERO2)
  target_compile_definitions(contrarebirth PRIVATE PI_ZERO2)
  target_compile_options(contrarebirth PRIVATE -mcpu=cortex-a53)
  # The integer mixer has independent sample loops that can use ARM NEON.
  set_source_files_properties(runtime/ax_dsp.cpp PROPERTIES COMPILE_OPTIONS "-O3")
  if(CMAKE_SIZEOF_VOID_P EQUAL 4)
    target_compile_options(contrarebirth PRIVATE -mfpu=neon-vfpv4 -mfloat-abi=hard)
    target_link_libraries(contrarebirth PRIVATE atomic)
  endif()
  if(NOT USE_GLES2)
    message(WARNING "Pi Zero 2 W hardware cannot run the OpenGL 3.3 renderer. Use -DUSE_GLES2=ON for hardware rendering.")
  endif()
endif()
# GCC 8 (Raspbian Buster) keeps C++17 filesystem in a separate library.
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
  target_link_libraries(contrarebirth PRIVATE stdc++fs)
endif()
if(WIN32)
  # SDL2 development package (SDL2-devel-*-mingw) location
  set(SDL2_DIR "" CACHE PATH "SDL2 mingw dev package root (x86_64-w64-mingw32 subfolder)")
  target_include_directories(contrarebirth PRIVATE runtime/gl ${SDL2_DIR}/include)
  target_compile_definitions(contrarebirth PRIVATE SDL_MAIN_HANDLED)
  target_compile_options(contrarebirth PRIVATE -O2 -fno-strict-aliasing -w -g0)
  target_link_directories(contrarebirth PRIVATE ${SDL2_DIR}/lib)
  target_sources(contrarebirth PRIVATE runtime/app.rc)
  target_link_libraries(contrarebirth PRIVATE SDL2 winmm -static-libgcc -static-libstdc++ -Wl,-Bstatic -lwinpthread -Wl,-Bdynamic)
elseif(USE_SDL)
  find_package(SDL2 REQUIRED)
  if(USE_GLES2)
    find_path(GLES2_INCLUDE_DIR GLES2/gl2.h HINTS /opt/vc/include)
    find_library(GLES2_LIBRARY NAMES brcmGLESv2 GLESv2 HINTS /opt/vc/lib)
    if(NOT GLES2_INCLUDE_DIR OR NOT GLES2_LIBRARY)
      message(FATAL_ERROR "OpenGL ES 2.0 development headers and library are required.")
    endif()
    target_include_directories(contrarebirth PRIVATE ${GLES2_INCLUDE_DIR})
    # Renderer selection only affects host runtime sources, not generated guest code.
    set_property(SOURCE ${RT_SRC} APPEND PROPERTY COMPILE_DEFINITIONS USE_GLES2)
    target_link_libraries(contrarebirth PRIVATE ${GLES2_LIBRARY})
  else()
    find_package(OpenGL REQUIRED)
    target_link_libraries(contrarebirth PRIVATE OpenGL::GL)
  endif()
  find_package(Threads REQUIRED)
  target_compile_definitions(contrarebirth PRIVATE USE_SDL)
  target_compile_options(contrarebirth PRIVATE -O2 -fno-strict-aliasing -w -g0)
  if(TARGET SDL2::SDL2)
    target_link_libraries(contrarebirth PRIVATE SDL2::SDL2)
  else()
    # Older SDL2 packages (including RetroPie/Buster) only export variables.
    target_include_directories(contrarebirth PRIVATE ${SDL2_INCLUDE_DIRS})
    target_link_libraries(contrarebirth PRIVATE ${SDL2_LIBRARIES})
  endif()
  target_link_libraries(contrarebirth PRIVATE Threads::Threads)
else()
  target_compile_options(contrarebirth PRIVATE -O2 -fno-strict-aliasing -w -g0)
  target_link_libraries(contrarebirth PRIVATE EGL GL pthread)
endif()
